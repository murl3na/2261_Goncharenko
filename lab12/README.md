### Задание 3 **(2)**

*Создайте два объекта* **x** *и* **y** *класса* **box** *в куче. Напишите
программу, для проверки работы методов, которая будет использовать
оба объекта. Не забывайте, что объекты созданные в куче следует
освобождать.*

```
#include <iostream>
class box {
private:
    double length;
    double width;
    double height;
public:
    box(double l = 1.0, double w = 1.0, double h = 1.0) 
        : length(l), width(w), height(h) {}
    double getVolume() const {
        return length * width * height;
    }
    void printInfo() const {
        std::cout << "Коробка [" << length << " x " << width << " x " << height 
                  << "], объём = " << getVolume() << std::endl;
    }
};
int main() {
    box* x = new box(2.0, 3.0, 4.0);
    box* y = new box(5.5, 2.0, 1.5);

    std::cout << "Объект x: ";
    x->printInfo();

    std::cout << "Объект y: ";
    y->printInfo();

    std::cout << "Суммарный объём: " << (x->getVolume() + y->getVolume()) << std::endl;

    delete x;
    delete y;

    x = nullptr;
    y = nullptr;
    return 0;
}
```
